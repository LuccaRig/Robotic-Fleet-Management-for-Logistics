cmake_minimum_required(VERSION 3.15)

add_library(communication_lib src/mqtt_client.cpp src/mqtt_integration.cpp src/protocol.cpp src/ipc_channel.cpp)
target_include_directories(communication_lib PUBLIC include ${PROJECT_SOURCE_DIR}/core/include)
find_package(Threads REQUIRED)
target_link_libraries(communication_lib PUBLIC Threads::Threads)

if(ENABLE_MQTT)
    find_path(MOSQUITTO_INCLUDE_DIR NAMES mosquitto.h)
    find_library(MOSQUITTO_LIB NAMES mosquitto)
    if(NOT MOSQUITTO_INCLUDE_DIR OR NOT MOSQUITTO_LIB)
        message(FATAL_ERROR "ENABLE_MQTT=ON exige os headers e a biblioteca libmosquitto. Instale a dependência ou use ENABLE_MQTT=OFF.")
    endif()
    find_path(CJSON_INCLUDE_DIR NAMES cJSON.h PATH_SUFFIXES cjson)
    find_library(CJSON_LIB NAMES cjson)
    if(NOT CJSON_INCLUDE_DIR OR NOT CJSON_LIB)
        message(FATAL_ERROR "ENABLE_MQTT=ON exige a biblioteca cJSON e seus headers para ler comandos JSON.")
    endif()
    target_include_directories(communication_lib PRIVATE ${MOSQUITTO_INCLUDE_DIR} ${CJSON_INCLUDE_DIR})
    target_link_libraries(communication_lib PRIVATE ${MOSQUITTO_LIB} ${CJSON_LIB})
    target_compile_definitions(communication_lib PUBLIC ENABLE_MQTT=1)
    message(STATUS "MQTT habilitado com libmosquitto")
endif()
